Experiment 1: String Operations Using Pointers
Question
Implement the string operations strcat(), strcmp(), and strcpy() using pointers.
Answer
#include <stdio.h>

void mystrcpy(char *d, char *s) {
    while ((*d++ = *s++) != '\0');
}

void mystrcat(char *d, char *s) {
    while (*d) d++;
    while ((*d++ = *s++) != '\0');
}

int mystrcmp(char *a, char *b) {
    while (*a && *a == *b) {
        a++;
        b++;
    }
    return *a - *b;
}

int main() {
    char s1[100] = "Hello ";
    char s2[] = "World";
    char s3[100];

    mystrcpy(s3, s2);
    printf("Copied: %s\n", s3);

    mystrcat(s1, s2);
    printf("Concatenated: %s\n", s1);

    printf("Comparison: %d\n",
           mystrcmp("Apple", "Apple"));

    return 0;
}

Output
Copied: World
Concatenated: Hello World
Comparison: 0

Steps
1. Declare character arrays to store the strings.
2. The mystrcpy() function copies the source string into the destination string using pointers.
3. The mystrcat() function moves to the end of the first string and appends the second string.
4. The mystrcmp() function compares two strings character by character.
5. If both strings are equal, mystrcmp() returns 0.
6. The main() function calls all three functions and displays their results.
Remember: strcpy() copies, strcat() joins, and strcmp() compares strings.

Experiment 3: Operations on Lengths Using Structures
Question
Implement length in feet and inches using a structure. Write routines to add, multiply, and subtract lengths.
Answer
#include <stdio.h>

struct Length {
    int feet, inches;
};

struct Length convert(int total) {
    struct Length l;
    l.feet = total / 12;
    l.inches = total % 12;
    return l;
}

struct Length add(struct Length a, struct Length b) {
    return convert((a.feet + b.feet) * 12
                   + a.inches + b.inches);
}

struct Length subtract(struct Length a, struct Length b) {
    return convert((a.feet - b.feet) * 12
                   + a.inches - b.inches);
}

struct Length multiply(struct Length a, int n) {
    return convert((a.feet * 12 + a.inches) * n);
}

void display(struct Length a) {
    printf("%d feet %d inches\n", a.feet, a.inches);
}

int main() {
    struct Length a = {5, 8}, b = {3, 9};

    printf("First length: ");
    display(a);

    printf("Second length: ");
    display(b);

    printf("Addition: ");
    display(add(a, b));

    printf("Subtraction: ");
    display(subtract(a, b));

    printf("Multiplication by 2: ");
    display(multiply(a, 2));

    return 0;
}

Output
First length: 5 feet 8 inches
Second length: 3 feet 9 inches
Addition: 9 feet 5 inches
Subtraction: 1 feet 11 inches
Multiplication by 2: 11 feet 4 inches

Steps
1. Define a structure containing feet and inches.
2. Convert each length into total inches to simplify calculations.
3. The add() function adds the total inches of both lengths.
4. The subtract() function subtracts the second length from the first.
5. The multiply() function multiplies a length by an integer.
6. The convert() function converts total inches back into feet and inches.
7. The display() function prints the result.
Remember: One foot equals 12 inches. Convert to inches before performing arithmetic, then convert back.

Experiment 4: Insert an Element at the kth Position
Question
Write a program to insert a new element into a given unsorted array at the kth position.
Answer
#include <stdio.h>

int main() {
    int a[100], n, k, value, i;

    printf("Enter array size: ");
    scanf("%d", &n);

    if (n < 0 || n >= 100) {
        printf("Invalid array size");
        return 0;
    }

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter position and value: ");
    scanf("%d %d", &k, &value);

    if (k < 1 || k > n + 1) {
        printf("Invalid position");
        return 0;
    }

    for (i = n; i >= k; i--)
        a[i] = a[i - 1];

    a[k - 1] = value;
    n++;

    printf("Array after insertion: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}

Output

Enter elements: 10 20 30
Enter position and value: 2 15
Array after insertion: 10 15 20 30

Steps
1. Declare an array and read its size and elements.
2. Read the insertion position k and the new value.
3. Check whether the position is valid and the array has enough space.
4. Shift elements one position to the right, starting from the last element.
5. Insert the new value at index k - 1.
6. Increase the array size by one.
7. Display the updated array.
Remember: Shift elements from right to left when inserting into an array.

Write a program to delete an element from a given array.

        printf("%d ", a[i]);

    return 0;
}

Array after deletion: 10 30 40

Steps
1. Read the array size and elements.
2. Read the value to delete.
3. Search the array for the first occurrence of that value.
4. Store its index in pos.
5. If the element is not found, display a message and stop.
6. Shift all elements after the deleted element one position to the left.
7. Decrease the array size and display the updated array.
Remember: Deletion shifts elements from left to right in the sense that each element is copied from its right-hand neighbor into the position before it.

Experiment 10: Stack Using an Array
Question
Write a program to implement a stack using an array, showing overflow and underflow in push and pop operations.
Answer
#include <stdio.h>
#define MAX 5

int stack[MAX], top = -1;

void push(int x) {
    if (top == MAX - 1)
        printf("Stack Overflow!\n");
    else {
        stack[++top] = x;
        printf("%d pushed\n", x);
    }
}

void pop() {
    if (top == -1)
        printf("Stack Underflow!\n");
    else
        printf("Popped: %d\n", stack[top--]);
}

void display() {
    int i;
    for (i = top; i >= 0; i--)
        printf("%d ", stack[i]);
    printf("\n");
}

int main() {
    push(10);
    push(20);
    push(30);
    display();
    pop();
    display();

    return 0;
}

Output
10 pushed
20 pushed
30 pushed
30 20 10
Popped: 30
20 10

Steps
1. Declare an array of size MAX to store stack elements.
2. Initialize top = -1 to indicate that the stack is empty.
3. The push() function checks whether the stack is full before inserting an element.
4. The pop() function checks whether the stack is empty before removing an element.
5. The display() function prints the elements from the top to the bottom.
6. The main() function demonstrates pushing, displaying, and popping elements.
Remember: A stack follows LIFO — Last In, First Out. Overflow occurs when inserting into a full stack, and underflow occurs when removing from an empty stack.
Experiment 11: Queue Using an Array
Question
Write a program to implement a queue using an array, showing insertion and deletion operations.
Answer
#include <stdio.h>
#define MAX 5

int q[MAX], front = 0, rear = -1;

void insert(int x) {
    if (rear == MAX - 1)
        printf("Queue Overflow!\n");
    else
        q[++rear] = x;
}

void delete() {
    if (front > rear)
        printf("Queue Underflow!\n");
    else
        printf("Deleted: %d\n", q[front++]);

    if (front > rear)
        front = 0, rear = -1;
}

void display() {
    int i;

    if (rear == -1) {
        printf("Queue is empty\n");
        return;
    }

    for (i = front; i <= rear; i++)
        printf("%d ", q[i]);

    printf("\n");
}

int main() {
    insert(10);
    insert(20);
    insert(30);
    display();
    delete();
    display();

    return 0;
}

Output
10 20 30
Deleted: 10
20 30

Steps
1. Declare an array and initialize front = 0 and rear = -1.
2. The insert() function adds a new element at the rear.
3. The delete() function removes an element from the front.
4. Check for overflow before insertion and underflow before deletion.
5. Reset the queue indices when the queue becomes empty.
6. Display the elements from the front to the rear.
Remember: A queue follows FIFO — First In, First Out. This is a simple linear queue; freed positions at the beginning are not reused until the queue becomes empty.
Experiment 12: Circular Queue Using an Array
Question
Write a program to implement a circular queue using an array, showing insertion and deletion operations.
Answer
#include <stdio.h>
#define MAX 5

int q[MAX], front = -1, rear = -1;

void insert(int x) {
    if ((rear + 1) % MAX == front) {
        printf("Queue Overflow!\n");
        return;
    }

    if (front == -1)
        front = 0;

    rear = (rear + 1) % MAX;
    q[rear] = x;

    printf("%d inserted\n", x);
}

void delete() {
    if (front == -1) {
        printf("Queue Underflow!\n");
        return;
    }

    printf("Deleted: %d\n", q[front]);

    if (front == rear)
        front = rear = -1;
    else
        front = (front + 1) % MAX;
}

void display() {
    int i;

    if (front == -1) {
        printf("Queue is empty\n");
        return;
    }

    i = front;

    while (1) {
        printf("%d ", q[i]);

        if (i == rear)
            break;

        i = (i + 1) % MAX;
    }

    printf("\n");
}

int main() {
    insert(10);
    insert(20);
    insert(30);
    display();
    delete();
    insert(40);
    display();

    return 0;
}

Output
10 inserted
20 inserted
30 inserted
10 20 30
Deleted: 10
40 inserted
20 30 40

Steps
1. Initialize front = -1 and rear = -1 to represent an empty queue.
2. Before insertion, check whether (rear + 1) % MAX == front. If true, the queue is full.
3. Move rear circularly using (rear + 1) % MAX and insert the element.
4. During deletion, remove the element at front.
5. If the deleted element was the last element, reset both indices to -1; otherwise, advance front circularly.
6. Display elements by moving through the array with modulo arithmetic.
Remember: The modulo operator % allows the indices to wrap around to the beginning of the array, reusing empty positions.

Experiment 13: Linear Linked List Operations
Question
Write a program to implement a Linear Linked List showing creation, display, insertion, deletion, and searching operations.
Answer
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;

void insert(int x, int pos) {
    struct Node *n = malloc(sizeof(struct Node));
    struct Node *t = head;
    int i;

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    if (pos < 1) {
        printf("Invalid position\n");
        free(n);
        return;
    }

    n->data = x;

    if (pos == 1) {
        n->next = head;
        head = n;
        return;
    }

    for (i = 1; t && i < pos - 1; i++)
        t = t->next;

    if (t == NULL) {
        printf("Invalid position\n");
        free(n);
        return;
    }

    n->next = t->next;
    t->next = n;
}

void deleteNode(int pos) {
    struct Node *t = head, *d;
    int i;

    if (pos < 1 || head == NULL) {
        printf("Invalid deletion\n");
        return;
    }

    if (pos == 1) {
        head = head->next;
        free(t);
        return;
    }

    for (i = 1; t && i < pos - 1; i++)
        t = t->next;

    if (t == NULL || t->next == NULL) {
        printf("Invalid position\n");
        return;
    }

    d = t->next;
    t->next = d->next;
    free(d);
}

void display() {
    struct Node *t = head;

    while (t) {
        printf("%d -> ", t->data);
        t = t->next;
    }

    printf("NULL\n");
}

void search(int x) {
    struct Node *t = head;
    int pos = 1;

    while (t) {
        if (t->data == x) {
            printf("Found at position %d\n", pos);
            return;
        }

        t = t->next;
        pos++;
    }

    printf("Element not found\n");
}

int main() {
    insert(10, 1);
    insert(20, 2);
    insert(30, 3);

    printf("Original list: ");
    display();

    insert(15, 2);
    printf("After insertion: ");
    display();

    deleteNode(3);
    printf("After deletion: ");
    display();

    search(30);

    return 0;
}

Output
Original list: 10 -> 20 -> 30 -> NULL
After insertion: 10 -> 15 -> 20 -> 30 -> NULL
After deletion: 10 -> 15 -> 30 -> NULL
Found at position 3

Steps
1. Define a node containing an integer data and a pointer next.
2. Use head to store the address of the first node.
3. The insert() function creates a node and links it at the specified position.
4. The deleteNode() function removes the node at the specified position and frees its memory.
5. The display() function traverses the list until it reaches NULL.
6. The search() function compares each node's data with the required value.
7. The main() function demonstrates creation, insertion, deletion, display, and searching.
Remember: In a singly linked list, each node points to the next node. The last node points to NULL.
Experiment 14: Stack Using a Linked List
Question
Write a program to implement a stack using a linked list with Push, Pop, and Display operations.
Answer
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *top = NULL;

void push(int x) {
    struct Node *n = malloc(sizeof(struct Node));

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    n->data = x;
    n->next = top;
    top = n;

    printf("%d pushed\n", x);
}

void pop() {
    struct Node *t;

    if (top == NULL) {
        printf("Stack Underflow!\n");
        return;
    }

    t = top;
    printf("Popped: %d\n", top->data);

    top = top->next;
    free(t);
}

void display() {
    struct Node *t = top;

    while (t) {
        printf("%d ", t->data);
        t = t->next;
    }

    printf("\n");
}

int main() {
    push(10);
    push(20);
    push(30);

    display();
    pop();
    display();

    return 0;
}

Output
10 pushed
20 pushed
30 pushed
30 20 10
Popped: 30
20 10

Steps
1. Create a node structure with data and next.
2. Initialize top = NULL to represent an empty stack.
3. In push(), create a new node and connect it to the current top.
4. Update top to point to the newly inserted node.
5. In pop(), check whether the stack is empty.
6. If it is not empty, remove the top node, update top, and free the removed node's memory.
7. Use display() to print elements from top to bottom.
Remember: The linked-list stack grows dynamically, so it does not have a fixed array capacity. It can still run out of memory.
Experiment 15: Queue Using a Linked List
Question
Write a program to implement a queue using a linked list with insertion, deletion, and display operations.
Answer
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *front = NULL, *rear = NULL;

void insert(int x) {
    struct Node *n = malloc(sizeof(struct Node));

        printf("Memory allocation failed\n");
        return;
    }

    n->data = x;
    n->next = NULL;

        cur = next;

}

#include <stdio.h>

void quickSort(int a[], int low, int high) {
    int i, j, pivot, temp;

    if (low >= high)
        return;

    pivot = a[low];
    i = low + 1;
    j = high;

    while (i <= j) {
        while (i <= high && a[i] <= pivot)
            i++;

        while (j > low && a[j] > pivot)
            j--;

        if (i < j) {
            temp = a[i];
            a[i] = a[j];
            a[j] = temp;
        }
    }

    a[low] = a[j];
    a[j] = pivot;

    quickSort(a, low, j - 1);
    quickSort(a, j + 1, high);
}

        printf("%d ", a[i]);

void merge(int a[], int low, int mid, int high) {
    int i = low, j = mid + 1, k = low;

    while (i <= mid && j <= high) {
        if (a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while (i <= mid)
        temp[k++] = a[i++];

    while (j <= high)
        temp[k++] = a[j++];

    for (i = low; i <= high; i++)
        a[i] = temp[i];
}

    }
}

Experiment 29: Breadth First Search (BFS)
Question
Write a program to implement Breadth First Search on a graph.
Answer
#include <stdio.h>

int graph[20][20], visited[20], queue[20], n;

void BFS(int start) {
    int front = 0, rear = 0;
    int v, i;

    visited[start] = 1;
    queue[rear++] = start;

    while (front < rear) {
        v = queue[front++];
        printf("%d ", v);

        for (i = 0; i < n; i++) {
            if (graph[v][i] == 1 && !visited[i]) {
                visited[i] = 1;
                queue[rear++] = i;
            }
        }
    }
}

Experiment 30: Prim's Algorithm for Minimum Spanning Tree
Question
Write a program to implement Prim's algorithm for a Minimum Spanning Tree (MST).
Answer
#include <stdio.h>

#define MAX 20
#define INF 999999

int main() {
    int graph[MAX][MAX], selected[MAX] = {0};
    int n, i, j, min, x, y;
    int edges = 0, cost = 0;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    if (n < 1 || n > MAX) {
        printf("Invalid number of vertices");
        return 0;
    }

    printf("Enter weighted adjacency matrix:\n");

    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &graph[i][j]);

    selected[0] = 1;

    printf("Edges in MST:\n");

    while (edges < n - 1) {
        min = INF;
        x = -1;
        y = -1;

        for (i = 0; i < n; i++) {
            if (selected[i]) {
                for (j = 0; j < n; j++) {
                    if (!selected[j] &&
                        graph[i][j] != 0 &&
                        graph[i][j] < min) {
                        min = graph[i][j];
                        x = i;
                        y = j;
                    }
                }
            }
        }

        if (x == -1) {
            printf("Graph is disconnected; MST does not exist.\n");
            return 0;
        }

        printf("%d - %d : Weight %d\n", x, y, min);

        cost += min;
        selected[y] = 1;
        edges++;
    }

    printf("Total cost of MST = %d\n", cost);

    return 0;
}

Enter weighted adjacency matrix:
0 10 6 5
10 0 0 15
6 0 0 4
5 15 4 0
Edges in MST:
0 - 3 : Weight 5
3 - 2 : Weight 4
0 - 1 : Weight 10
Total cost of MST = 19

Steps
1. Read the number of vertices and the weighted adjacency matrix.
2. Mark vertex 0 as selected to start building the MST.
3. Find the minimum-weight edge connecting a selected vertex to an unselected vertex.
4. Print the selected edge and add its weight to the total cost.
5. Mark the newly connected vertex as selected.
6. Repeat until n - 1 edges have been selected.
7. If no connecting edge exists before the MST is complete, report that the graph is disconnected.
8. Display the total cost of the MST.
Remember: Prim's algorithm repeatedly adds the minimum-weight edge that connects the existing tree to a new vertex. This implementation assumes an undirected, connected graph with nonnegative edge weights; a matrix entry of 0 represents no edge.
