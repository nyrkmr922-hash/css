Q1. Implement the string operations strcat(), strcmp(), and strcpy() using pointers.
#include <stdio.h>

void mystrcpy(char *dest, char *src) {
    while ((*dest++ = *src++) != '\0');
}

void mystrcat(char *dest, char *src) {
    while (*dest != '\0')
        dest++;

    while ((*dest++ = *src++) != '\0');
}

int mystrcmp(char *s1, char *s2) {
    while (*s1 && *s2 && *s1 == *s2) {
        s1++;
        s2++;
    }
    return *s1 - *s2;
}

int main() {
    char s1[100] = "Hello ";
    char s2[] = "World";
    char s3[100];

    mystrcpy(s3, s2);
    printf("Copied string: %s\n", s3);

    mystrcat(s1, s2);
    printf("Concatenated string: %s\n", s1);

    printf("Comparison result: %d\n",
           mystrcmp("Apple", "Apple"));

    return 0;
}

Q3. Implement length in feet and inches using a structure. Write routines to add, multiply, and subtract lengths.
#include <stdio.h>

struct Length {
    int feet;
    int inches;
};

struct Length normalize(int feet, int inches) {
    struct Length l;

    feet += inches / 12;
    inches %= 12;

    if (inches < 0) {
        inches += 12;
        feet--;
    }

    l.feet = feet;
    l.inches = inches;

    return l;
}

struct Length add(struct Length a, struct Length b) {
    return normalize(a.feet + b.feet,
                     a.inches + b.inches);
}

struct Length subtract(struct Length a, struct Length b) {
    return normalize(a.feet - b.feet,
                     a.inches - b.inches);
}

struct Length multiply(struct Length a, int n) {
    int total = (a.feet * 12 + a.inches) * n;
    return normalize(total / 12, total % 12);
}

void display(struct Length l) {
    printf("%d feet %d inches\n", l.feet, l.inches);
}

int main() {
    struct Length a = {5, 8};
    struct Length b = {3, 9};

    printf("First length: ");
    display(a);

    printf("Second length: ");
    display(b);

    printf("Addition: ");
    display(add(a, b));

    printf("Subtraction: ");
    display(subtract(a, b));

    printf("Multiplication of first length by 2: ");
    display(multiply(a, 2));

    return 0;
}

Q1. Implement Stack using an array, showing overflow and underflow in push and pop operations.
#include <stdio.h>
#define MAX 5

int stack[MAX];
int top = -1;

void push(int value) {
    if (top == MAX - 1) {
        printf("Stack Overflow!\n");
        return;
    }

    stack[++top] = value;
    printf("%d pushed\n", value);
}

void pop() {
    if (top == -1) {
        printf("Stack Underflow!\n");
        return;
    }

    printf("Popped: %d\n", stack[top--]);
}

void display() {
    int i;

    if (top == -1) {
        printf("Stack is empty\n");
        return;
    }

    for (i = top; i >= 0; i--)
        printf("%d ", stack[i]);

    printf("\n");
}

int main() {
    push(10);
    push(20);
    push(30);

    display();

    pop();
    display();

    return 0;
}

Q3. Implement Circular Queue using an array, showing insertion and deletion operations.
#include <stdio.h>
#define MAX 5

int queue[MAX];
int front = -1, rear = -1;

void insert(int value) {
    if ((rear + 1) % MAX == front) {
        printf("Circular Queue Overflow!\n");
        return;
    }

    if (front == -1)
        front = 0;

    rear = (rear + 1) % MAX;
    queue[rear] = value;

    printf("%d inserted\n", value);
}

void delete() {
    if (front == -1) {
        printf("Circular Queue Underflow!\n");
        return;
    }

    printf("Deleted: %d\n", queue[front]);

    if (front == rear) {
        front = rear = -1;
    } else {
        front = (front + 1) % MAX;
    }
}

void display() {
    int i;

    if (front == -1) {
        printf("Queue is empty\n");
        return;
    }

    i = front;

    while (1) {
        printf("%d ", queue[i]);

        if (i == rear)
            break;

        i = (i + 1) % MAX;
    }

    printf("\n");
}

int main() {
    insert(10);
    insert(20);
    insert(30);

    display();

    delete();
    insert(40);

    display();

    return 0;

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;

void create(int value) {
    struct Node *n = malloc(sizeof(struct Node));
    struct Node *temp;

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    n->data = value;
    n->next = NULL;

    if (head == NULL) {
        head = n;
        return;
    }

    temp = head;

    while (temp->next != NULL)
        temp = temp->next;

    temp->next = n;
}

void display() {
    struct Node *temp = head;

    while (temp != NULL) {
        printf("%d -> ", temp->data);
        temp = temp->next;
    }

    printf("NULL\n");
}

void insert(int value, int position) {
    struct Node *n, *temp;
    int i;

    n = malloc(sizeof(struct Node));

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    n->data = value;

    if (position == 1) {
        n->next = head;
        head = n;
        return;
    }

    temp = head;

    for (i = 1; temp != NULL && i < position - 1; i++)
        temp = temp->next;

    if (temp == NULL) {
        printf("Invalid position\n");
        free(n);
        return;
    }

    n->next = temp->next;
    temp->next = n;
}

void deleteNode(int position) {
    struct Node *temp, *del;
    int i;

    if (head == NULL || position < 1) {
        printf("Invalid deletion\n");
        return;
    }

    if (position == 1) {
        del = head;
        head = head->next;
        free(del);
        return;
    }

    temp = head;

    for (i = 1; temp != NULL && i < position - 1; i++)
        temp = temp->next;

    if (temp == NULL || temp->next == NULL) {
        printf("Invalid position\n");
        return;
    }

    del = temp->next;
    temp->next = del->next;
    free(del);
}

void search(int value) {
    struct Node *temp = head;
    int position = 1;

    while (temp != NULL) {
        if (temp->data == value) {
            printf("Found at position %d\n", position);
            return;
        }

        temp = temp->next;
        position++;
    }

    printf("Element not found\n");
}

void reverse() {
    struct Node *prev = NULL;
    struct Node *current = head;
    struct Node *next;

    while (current != NULL) {
        next = current->next;
        current->next = prev;
        prev = current;
        current = next;
    }

    head = prev;
}

        printf("%d -> ", temp->data);
        temp = temp->next;

Q3. Implement Doubly Linked List (creation, display, insertion, deletion, searching).
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *prev;
    struct Node *next;
};

struct Node *head = NULL;

void insert(int value, int position) {
    struct Node *n, *temp;
    int i;

    if (position < 1) {
        printf("Invalid position\n");
        return;
    }

    n = malloc(sizeof(struct Node));

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    n->data = value;
    n->prev = NULL;
    n->next = NULL;

    if (position == 1) {
        n->next = head;

        if (head != NULL)
            head->prev = n;

        head = n;
        return;
    }

    temp = head;

    for (i = 1; temp != NULL && i < position - 1; i++)
        temp = temp->next;

    if (temp == NULL) {
        printf("Invalid position\n");
        free(n);
        return;
    }

    n->next = temp->next;
    n->prev = temp;

    if (temp->next != NULL)
        temp->next->prev = n;

    temp->next = n;
}

void deleteNode(int position) {
    struct Node *temp;
    int i;

    if (position < 1 || head == NULL) {
        printf("Invalid deletion\n");
        return;
    }

    temp = head;

    for (i = 1; temp != NULL && i < position; i++)
        temp = temp->next;

    if (temp == NULL) {
        printf("Invalid position\n");
        return;
    }

    if (temp->prev != NULL)
        temp->prev->next = temp->next;
    else
        head = temp->next;

    if (temp->next != NULL)
        temp->next->prev = temp->prev;

    free(temp);
}

}

void display() {
    struct Node *temp = head;

    while (temp != NULL) {
        printf("%d <-> ", temp->data);
        temp = temp->next;
    }

    printf("NULL\n");
}

Q1. Create a BST and display using recursive Preorder, Inorder, and Postorder traversal.
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *insert(struct Node *root, int value) {
    if (root == NULL) {
        struct Node *n = malloc(sizeof(struct Node));

        if (n == NULL) {
            printf("Memory allocation failed\n");
            return NULL;
        }

        n->data = value;
        n->left = n->right = NULL;

        return n;
    }

    if (value < root->data)
        root->left = insert(root->left, value);
    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

        n->left = n->right = NULL;

    }

    if (value < root->data)
        root->left = insert(root->left, value);
    else if (value > root->data)
        root->right = insert(root->right, value);

}

    while (root != NULL && root->left != NULL)
        root = root->left;

    return root;
}

struct Node *deleteNode(struct Node *root, int value) {
    struct Node *temp;

    if (root == NULL)
        return NULL;

    if (value < root->data) {
        root->left = deleteNode(root->left, value);
    } else if (value > root->data) {
        root->right = deleteNode(root->right, value);
    } else {
        if (root->left == NULL) {
            temp = root->right;
            free(root);
            return temp;
        }

        if (root->right == NULL) {
            temp = root->left;
            free(root);
            return temp;
        }

        temp = findMin(root->right);
        root->data = temp->data;
        root->right = deleteNode(root->right, temp->data);
    }

    return root;
}

Q2. Sort a given array using Quick Sort.
#include <stdio.h>

void quickSort(int a[], int low, int high) {
    int i, j, pivot, temp;

    if (low >= high)
        return;

    pivot = a[low];
    i = low + 1;
    j = high;

    while (i <= j) {
        while (i <= high && a[i] <= pivot)
            i++;

        while (a[j] > pivot)
            j--;

        if (i < j) {
            temp = a[i];
            a[i] = a[j];
            a[j] = temp;
        }
    }

    a[low] = a[j];
    a[j] = pivot;

    quickSort(a, low, j - 1);
    quickSort(a, j + 1, high);
}

    scanf("%d", &n);

        scanf("%d", &a[i]);

void merge(int a[], int low, int mid, int high) {
    int i = low, j = mid + 1, k = low;

    while (i <= mid && j <= high) {
        if (a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while (i <= mid)
        temp[k++] = a[i++];

    while (j <= high)
        temp[k++] = a[j++];

    for (i = low; i <= high; i++)
        a[i] = temp[i];
}

    }
}

int main() {

    int selected[MAX] = {0};
    int n, i, j;
    int edges = 0;
    int totalCost = 0;
    int min, x = -1, y = -1;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    if (n < 1 || n > MAX) {
        printf("Invalid number of vertices\n");
        return 0;
    }

    printf("Enter weighted adjacency matrix:\n");
    printf("Enter 0 where no edge exists.\n");

    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++)
            scanf("%d", &graph[i][j]);
    }

    selected[0] = 1;

    printf("Edges in Minimum Spanning Tree:\n");

    while (edges < n - 1) {
        min = INF;
        x = -1;
        y = -1;

        for (i = 0; i < n; i++) {
            if (selected[i]) {
                for (j = 0; j < n; j++) {
                    if (!selected[j] &&
                        graph[i][j] != 0 &&
                        graph[i][j] < min) {

                        min = graph[i][j];
                        x = i;
                        y = j;
                    }
                }
            }
        }

        if (x == -1) {
            printf("Graph is disconnected; MST does not exist.\n");
            return 0;
        }

        printf("%d - %d : Weight %d\n", x, y, min);

        totalCost += min;
        selected[y] = 1;
        edges++;
    }

    printf("Total cost of MST = %d\n", totalCost);

    return 0;
}
